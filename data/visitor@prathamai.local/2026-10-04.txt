});

const joystickZone = document.getElementById('joystick-zone');

function updateKnob(cx, cy) {
  const dx = cx - touchOrigin.x;
  const dy = cy - touchOrigin.y;
  const dist = Math.hypot(dx, dy);
  const maxR = 45;
  const angle = Math.atan2(dy, dx);
  const clampedDist = Math.min(dist, maxR);
  const kx = Math.cos(angle) * clampedDist;
  const ky = Math.sin(angle) * clampedDist;
  joystickKnob.style.transform = `translate(${kx}px, ${ky}px)`;
  joyMove = { x: kx / maxR, y: ky / maxR };
}

const GRAVITY = 0.58;

function checkGround(pos) {
  for (const col of colliders) {
    const b = col.box;
    if (pos.x >= b.min.x - 0.9 && pos.x <= b.max.x + 0.9 &&
        pos.z >= b.min.z - 0.9 && pos.z <= b.max.z + 0.9) {
      if (pos.y >= col.topY - 0.5 && pos.y <= col.topY + 1.2) {
        return col.topY;
      }
    }
  }
  return null;
}

function respawnGuy(guy, defaultZ = 20) {
  guy.pos.set((Math.random() - 0.5) * 8, 14, Math.min(guy.pos.z + 18, defaultZ));
  guy.vel.set(0, 0, 0);
  guy.stumbleTimer = 25;
  guy.isDiving = false;
}

const particles = [];
function spawnPopParticles(x, y, z, color = 0xffeaa7) {
  for (let i = 0; i < 8; i++) {
    const pGeo = new THREE.SphereGeometry(0.35, 6, 6);
    const pMat = new THREE.MeshBasicMaterial({ color });
    const pMesh = new THREE.Mesh(pGeo, pMat);
    pMesh.position.set(x, y, z);
    scene.add(pMesh);
    particles.push({
      mesh: pMesh,
      vx: (Math.random() - 0.5) * 0.7,
      vy: Math.random() * 0.6 + 0.2,
      vz: (Math.random() - 0.5) * 0.7,
      life: 25
    });
  }
}

let raceStartTime = Date.now();
let raceRunning = true;

function updatePhysics() {
  const time = (Date.now() - raceStartTime) * 0.001;

  dynamicObstacles.forEach(obs => {
    if (obs.type === 'rotator') obs.mesh.rotation.y += obs.speed;
    else if (obs.type === 'roller') obs.mesh.rotation.x += obs.speed;
    else if (obs.type === 'pendulum') obs.mesh.rotation.z = Math.sin(time * obs.speed + obs.timeOffset) * obs.amplitude;
  });

  if (player.stumbleTimer > 0) {
    player.stumbleTimer--;
    player.model.group.rotation.x += 0.2;
    player.model.group.rotation.z += 0.1;
  } else {
    let moveX = 0, moveZ = 0;
    if (keys.w) moveZ -= 1;
    if (keys.s) moveZ += 1;
    if (keys.a) moveX -= 1;
    if (keys.d) moveX += 1;
    if (touchActive) { moveX += joyMove.x; moveZ += joyMove.y; }

    const inputLen = Math.hypot(moveX, moveZ);
    if (inputLen > 0.05) {
      const normX = moveX / Math.max(1, inputLen);
      const normZ = moveZ / Math.max(1, inputLen);
      const curSpeed = player.isDiving ? player.speed * 1.55 : player.speed;

      player.vel.x = normX * curSpeed;
      player.vel.z = normZ * curSpeed;

      const targetAngle = Math.atan2(normX, normZ) + Math.PI;
      player.model.group.rotation.y = targetAngle;
      player.model.group.rotation.x = player.isDiving ? Math.PI / 2.6 : 0;
      player.model.group.rotation.z = 0;
      player.animTime += 0.25;
    } else {
      player.vel.x *= 0.75;
      player.vel.z *= 0.75;
      if (!player.isDiving) player.model.group.rotation.x = 0;
    }

    if (keys.jump && player.isGrounded) {
      player.vel.y = 7.8;
      player.isGrounded = false;
      spawnPopParticles(player.pos.x, player.pos.y, player.pos.z, 0xffffff);
    }

    if (keys.dive && !player.isDiving && !player.isGrounded) {
      player.isDiving = true;
      player.vel.y = 2.5;
      player.vel.z -= 0.6;
      spawnPopParticles(player.pos.x, player.pos.y, player.pos.z, 0xfeca57);
    }
  }

  player.vel.y -= GRAVITY;
  player.pos.x += player.vel.x;
  player.pos.y += player.vel.y * 0.1;
  player.pos.z += player.vel.z;

  const gY = checkGround(player.pos);
  if (gY !== null && player.vel.y <= 0) {
    player.pos.y = gY;
    player.vel.y = 0;
    player.isGrounded = true;
    if (player.isDiving) {
      player.isDiving = false;
      player.stumbleTimer = 10;
    }
  } else {
    player.isGrounded = false;
  }

  if (player.pos.y < -15) respawnGuy(player);
  checkObstacleCollisions(player);

  player.model.group.position.copy(player.pos);
  animateGuyWalk(player);

  aiRacers.forEach(ai => {
    if (ai.stumbleTimer > 0) {
      ai.stumbleTimer--;
    } else {
      let dirZ = -1, dirX = 0;
      if (ai.pos.x < -8) dirX = 0.6;
      else if (ai.pos.x > 8) dirX = -0.6;
      else dirX = (Math.sin(time + ai.speed * 10) * 0.4);

      ai.vel.x = dirX * ai.speed;
      ai.vel.z = dirZ * ai.speed;
      ai.model.group.rotation.y = Math.atan2(dirX, dirZ) + Math.PI;

      ai.jumpCooldown--;
      if (ai.jumpCooldown <= 0 && ai.isGrounded) {
        if (Math.random() < 0.45) {
          ai.vel.y = 7.5;
          ai.isGrounded = false;
        }
        ai.jumpCooldown = 40 + Math.random() * 80;
      }
    }

    ai.vel.y -= GRAVITY;
    ai.pos.x += ai.vel.x;
    ai.pos.y += ai.vel.y * 0.1;
    ai.pos.z += ai.vel.z;

    const botGY = checkGround(ai.pos);
    if (botGY !== null && ai.vel.y <= 0) {
      ai.pos.y = botGY;
      ai.vel.y = 0;
      ai.isGrounded = true;
    } else {
      ai.isGrounded = false;
    }

    if (ai.pos.y < -15) respawnGuy(ai, 15);
    checkObstacleCollisions(ai);
    ai.model.group.position.copy(ai.pos);
    ai.animTime += 0.22;
    animateGuyWalk(ai);
  });

  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.mesh.position.x += p.vx;
    p.mesh.position.y += p.vy;
    p.mesh.position.z += p.vz;
    p.vy -= 0.02;
    p.life--;
    if (p.life <= 0) {
      scene.remove(p.mesh);
      particles.splice(i, 1);
    }
  }

  const allRacers = [player, ...aiRacers];
  allRacers.sort((a, b) => a.pos.z - b.pos.z);
  const playerRank = allRacers.indexOf(player) + 1;
  document.getElementById('rank-display').innerText = `POSITION: ${playerRank} / ${allRacers.length}`;

  if (raceRunning) {
    const elapsed = ((Date.now() - raceStartTime) * 0.001).toFixed(1);
    document.getElementById('timer-display').innerText = `TIME: ${elapsed}s`;

    if (player.pos.z <= FINISH_Z && !player.finished) {
      player.finished = true;
      raceRunning = false;
      showVictory(playerRank);
    }
  }

  const targetCamX = player.pos.x * 0.7;
  const targetCamY = player.pos.y + 7.5;
  const targetCamZ = player.pos.z + 16.5;

  camera.position.x += (targetCamX - camera.position.x) * 0.12;
  camera.position.y += (targetCamY - camera.position.y) * 0.12;
  camera.position.z += (targetCamZ - camera.position.z) * 0.12;
  camera.lookAt(player.pos.x, player.pos.y + 2.5, player.pos.z - 4);
}

function animateGuyWalk(guy) {
  if (guy.isGrounded && guy.stumbleTimer <= 0 && (Math.abs(guy.vel.x) > 0.05 || Math.abs(guy.vel.z) > 0.05)) {
    const legSwing = Math.sin(guy.animTime * 3) * 0.6;
    guy.model.leftLeg.rotation.x = legSwing;
    guy.model.rightLeg.rotation.x = -legSwing;
    guy.model.leftArm.rotation.x = -legSwing * 0.8;
    guy.model.rightArm.rotation.x = legSwing * 0.8;
    guy.model.group.rotation.z = Math.sin(guy.animTime * 3) * 0.12;
  } else {
    guy.model.leftLeg.rotation.x = 0;
    guy.model.rightLeg.rotation.x = 0;
    guy.model.leftArm.rotation.x = 0;
    guy.model.rightArm.rotation.x = 0;
  }
}

function checkObstacleCollisions(guy) {
  dynamicObstacles.forEach(obs => {
    if (obs.type === 'rotator') {
      const dZ = Math.abs(guy.pos.z - obs.center.z);
      const dX = Math.abs(guy.pos.x - obs.center.x);
      if (dZ < 2.5 && dX < obs.radius && Math.abs(guy.pos.y - obs.center.y) < 3.2) {
        const hitAngle = obs.mesh.rotation.y;
        guy.vel.x += Math.cos(hitAngle) * 1.5;
        guy.vel.z += Math.sin(hitAngle) * 1.5;
        guy.vel.y = 8;
        guy.stumbleTimer = 30;
        spawnPopParticles(guy.pos.x, guy.pos.y + 2, guy.pos.z, 0xff4757);
      }
    } else if (obs.type === 'roller') {
      const dist = Math.hypot(guy.pos.x - obs.center.x, guy.pos.z - obs.center.z);
      if (dist < 4.2 && Math.abs(guy.pos.y - obs.center.y) < 4.0) {
        guy.vel.z += 1.2;
        guy.vel.y = 6;
        guy.stumbleTimer = 25;
        spawnPopParticles(guy.pos.x, guy.pos.y + 2, guy.pos.z, 0x9b59b6);
      }
    } else if (obs.type === 'pendulum') {
      const headWorldPos = new THREE.Vector3();
      obs.head.getWorldPosition(headWorldPos);
      const dist = guy.pos.distanceTo(headWorldPos);
      if (dist < 4.5) {
        const knockX = (guy.pos.x - headWorldPos.x) > 0 ? 1.6 : -1.6;
        guy.vel.x = knockX;
        guy.vel.y = 9;
        guy.vel.z += 0.8;
        guy.stumbleTimer = 35;
        spawnPopParticles(guy.pos.x, guy.pos.y + 2, guy.pos.z, 0xee5253);
      }
    }
  });
}

function showVictory(rank) {
  const modal = document.getElementById('end-modal');

  const desc = document.getElementById('end-desc');
  const banner = document.getElementById('hud-banner');

    title.innerText = "ELIMINATED! 💥";
    desc.innerText = `Rank ${rank}. Try again to claim the crown!`;
  }
  modal.style.display = 'block';
}

function restartRace() {

    ai.vel.set(0, 0, 0);
    ai.finished = false;
    ai.stumbleTimer = 0;
  });
}
