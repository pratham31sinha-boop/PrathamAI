function updateKnob(cx, cy) {
  const dx = cx - touchOrigin.x;
  const dy = cy - touchOrigin.y;
  const dist = Math.hypot(dx, dy);
  const maxR = 45;
  const angle = Math.atan2(dy, dx);
  const clampedDist = Math.min(dist, maxR);
  const kx = Math.cos(angle) * clampedDist;
  const ky = Math.sin(angle) * clampedDist;
  joystickKnob.style.transform = `translate(${kx}px, ${ky}px)`;
  joyMove = { x: kx / maxR, y: ky / maxR };
}

function checkGround(pos) {
  for (const col of colliders) {
    const b = col.box;
    if (pos.x >= b.min.x - 0.9 && pos.x <= b.max.x + 0.9 &&
        pos.z >= b.min.z - 0.9 && pos.z <= b.max.z + 0.9) {
      if (pos.y >= col.topY - 0.5 && pos.y <= col.topY + 1.2) {
        return col.topY;
      }
    }
  }
  return null;
}

function respawnGuy(guy, defaultZ = 20) {

  guy.vel.set(0, 0, 0);
  guy.stumbleTimer = 25;

function spawnPopParticles(x, y, z, color = 0xffeaa7) {
  for (let i = 0; i < 8; i++) {

let raceStartTime = Date.now();

function updatePhysics() {
  const time = (Date.now() - raceStartTime) * 0.001;

  dynamicObstacles.forEach(obs => {
    if (obs.type === 'rotator') obs.mesh.rotation.y += obs.speed;
    else if (obs.type === 'roller') obs.mesh.rotation.x += obs.speed;
    else if (obs.type === 'pendulum') obs.mesh.rotation.z = Math.sin(time * obs.speed + obs.timeOffset) * obs.amplitude;
  });

  if (player.stumbleTimer > 0) {
    player.stumbleTimer--;
    player.model.group.rotation.x += 0.2;
    player.model.group.rotation.z += 0.1;
  } else {
    let moveX = 0, moveZ = 0;
    if (keys.w) moveZ -= 1;
    if (keys.s) moveZ += 1;
    if (keys.a) moveX -= 1;
    if (keys.d) moveX += 1;
    if (touchActive) { moveX += joyMove.x; moveZ += joyMove.y; }

    const inputLen = Math.hypot(moveX, moveZ);
    if (inputLen > 0.05) {
      const normX = moveX / Math.max(1, inputLen);
      const normZ = moveZ / Math.max(1, inputLen);
      const curSpeed = player.isDiving ? player.speed * 1.55 : player.speed;

      player.vel.x = normX * curSpeed;
      player.vel.z = normZ * curSpeed;

      const targetAngle = Math.atan2(normX, normZ) + Math.PI;
      player.model.group.rotation.y = targetAngle;
      player.model.group.rotation.x = player.isDiving ? Math.PI / 2.6 : 0;
      player.model.group.rotation.z = 0;
      player.animTime += 0.25;
    } else {
      player.vel.x *= 0.75;
      player.vel.z *= 0.75;
      if (!player.isDiving) player.model.group.rotation.x = 0;
    }

    if (keys.jump && player.isGrounded) {
      player.vel.y = 7.8;
      player.isGrounded = false;
      spawnPopParticles(player.pos.x, player.pos.y, player.pos.z, 0xffffff);
    }

    if (keys.dive && !player.isDiving && !player.isGrounded) {
      player.isDiving = true;
      player.vel.y = 2.5;
      player.vel.z -= 0.6;
      spawnPopParticles(player.pos.x, player.pos.y, player.pos.z, 0xfeca57);
    }
  }

  player.vel.y -= GRAVITY;
  player.pos.x += player.vel.x;
  player.pos.y += player.vel.y * 0.1;
  player.pos.z += player.vel.z;

  const gY = checkGround(player.pos);
  if (gY !== null && player.vel.y <= 0) {
    player.pos.y = gY;
    player.vel.y = 0;
    player.isGrounded = true;
    if (player.isDiving) {
      player.isDiving = false;
      player.stumbleTimer = 10;
    }
  } else {
    player.isGrounded = false;
  }

  if (player.pos.y < -15) respawnGuy(player);
  checkObstacleCollisions(player);

  player.model.group.position.copy(player.pos);
  animateGuyWalk(player);

  aiRacers.forEach(ai => {
    if (ai.stumbleTimer > 0) {
      ai.stumbleTimer--;
    } else {
      let dirZ = -1, dirX = 0;
      if (ai.pos.x < -8) dirX = 0.6;
      else if (ai.pos.x > 8) dirX = -0.6;
      else dirX = (Math.sin(time + ai.speed * 10) * 0.4);

      ai.vel.x = dirX * ai.speed;
      ai.vel.z = dirZ * ai.speed;
      ai.model.group.rotation.y = Math.atan2(dirX, dirZ) + Math.PI;

      ai.jumpCooldown--;
      if (ai.jumpCooldown <= 0 && ai.isGrounded) {
        if (Math.random() < 0.45) {
          ai.vel.y = 7.5;
          ai.isGrounded = false;
        }
        ai.jumpCooldown = 40 + Math.random() * 80;
      }
    }

    ai.vel.y -= GRAVITY;
    ai.pos.x += ai.vel.x;
    ai.pos.y += ai.vel.y * 0.1;
    ai.pos.z += ai.vel.z;

    const botGY = checkGround(ai.pos);
    if (botGY !== null && ai.vel.y <= 0) {
      ai.pos.y = botGY;
      ai.vel.y = 0;
      ai.isGrounded = true;
    } else {
      ai.isGrounded = false;
    }

    if (ai.pos.y < -15) respawnGuy(ai, 15);
    checkObstacleCollisions(ai);
    ai.model.group.position.copy(ai.pos);
    ai.animTime += 0.22;
    animateGuyWalk(ai);
  });

  for (let i = particles.length - 1; i >= 0; i--) {
    const p = particles[i];
    p.mesh.position.x += p.vx;
    p.mesh.position.y += p.vy;
    p.mesh.position.z += p.vz;
    p.vy -= 0.02;
    p.life--;
    if (p.life <= 0) {
      scene.remove(p.mesh);
      particles.splice(i, 1);
    }
  }

  const allRacers = [player, ...aiRacers];
  allRacers.sort((a, b) => a.pos.z - b.pos.z);
  const playerRank = allRacers.indexOf(player) + 1;
  document.getElementById('rank-display').innerText = `POSITION: ${playerRank} / ${allRacers.length}`;

  if (raceRunning) {
    const elapsed = ((Date.now() - raceStartTime) * 0.001).toFixed(1);
    document.getElementById('timer-display').innerText = `TIME: ${elapsed}s`;

    if (player.pos.z <= FINISH_Z && !player.finished) {
      player.finished = true;
      raceRunning = false;
      showVictory(playerRank);
    }
  }

  const targetCamX = player.pos.x * 0.7;
  const targetCamY = player.pos.y + 7.5;
  const targetCamZ = player.pos.z + 16.5;

  camera.position.x += (targetCamX - camera.position.x) * 0.12;
  camera.position.y += (targetCamY - camera.position.y) * 0.12;
  camera.position.z += (targetCamZ - camera.position.z) * 0.12;
  camera.lookAt(player.pos.x, player.pos.y + 2.5, player.pos.z - 4);
}

function animateGuyWalk(guy) {
  if (guy.isGrounded && guy.stumbleTimer <= 0 && (Math.abs(guy.vel.x) > 0.05 || Math.abs(guy.vel.z) > 0.05)) {
    const legSwing = Math.sin(guy.animTime * 3) * 0.6;
    guy.model.leftLeg.rotation.x = legSwing;
    guy.model.rightLeg.rotation.x = -legSwing;
    guy.model.leftArm.rotation.x = -legSwing * 0.8;
    guy.model.rightArm.rotation.x = legSwing * 0.8;
    guy.model.group.rotation.z = Math.sin(guy.animTime * 3) * 0.12;

    guy.model.leftLeg.rotation.x = 0;
    guy.model.rightLeg.rotation.x = 0;
    guy.model.leftArm.rotation.x = 0;
    guy.model.rightArm.rotation.x = 0;
  }
}

function checkObstacleCollisions(guy) {
  dynamicObstacles.forEach(obs => {
    if (obs.type === 'rotator') {
      const dZ = Math.abs(guy.pos.z - obs.center.z);
      const dX = Math.abs(guy.pos.x - obs.center.x);
      if (dZ < 2.5 && dX < obs.radius && Math.abs(guy.pos.y - obs.center.y) < 3.2) {
        const hitAngle = obs.mesh.rotation.y;
        guy.vel.x += Math.cos(hitAngle) * 1.5;
        guy.vel.z += Math.sin(hitAngle) * 1.5;
        guy.vel.y = 8;
        guy.stumbleTimer = 30;
        spawnPopParticles(guy.pos.x, guy.pos.y + 2, guy.pos.z, 0xff4757);
      }
    } else if (obs.type === 'roller') {
      const dist = Math.hypot(guy.pos.x - obs.center.x, guy.pos.z - obs.center.z);
      if (dist < 4.2 && Math.abs(guy.pos.y - obs.center.y) < 4.0) {
        guy.vel.z += 1.2;
        guy.vel.y = 6;
        guy.stumbleTimer = 25;
        spawnPopParticles(guy.pos.x, guy.pos.y + 2, guy.pos.z, 0x9b59b6);
      }
    } else if (obs.type === 'pendulum') {
      const headWorldPos = new THREE.Vector3();
      obs.head.getWorldPosition(headWorldPos);
      const dist = guy.pos.distanceTo(headWorldPos);
      if (dist < 4.5) {
        const knockX = (guy.pos.x - headWorldPos.x) > 0 ? 1.6 : -1.6;
        guy.vel.x = knockX;
        guy.vel.y = 9;
        guy.vel.z += 0.8;
        guy.stumbleTimer = 35;
        spawnPopParticles(guy.pos.x, guy.pos.y + 2, guy.pos.z, 0xee5253);
      }
    }
  });
}

class Terrain {
  constructor() {
    this.step = 16;
  }

  getHeight(x) {
    if (x < 150) return 480;
    const h1 = Math.sin(x * 0.002) * 110;
    const h2 = Math.cos(x * 0.0055) * 65;
    const h3 = Math.sin(x * 0.013) * 30;
    const h4 = Math.sin(x * 0.035) * 8;
    const ramp = Math.min((x - 150) * 0.05, 40);
    return 480 - (h1 + h2 + h3 + h4 + ramp);
  }

  getNormal(x) {
    const dx = 1.0;
    const y1 = this.getHeight(x - dx);
    const y2 = this.getHeight(x + dx);
    const angle = Math.atan2(y2 - y1, dx * 2);
    return {
      angle: angle,
      nx: -Math.sin(angle),
      ny: Math.cos(angle)
    };
  }
}

// --- PARTICLE EMITTER ---
class ParticleSystem {
  constructor() {
    this.particles = [];
  }
  add(x, y, vx, vy, color, size, life, type = 'circle') {
    this.particles.push({ x, y, vx, vy, color, size, initialSize: size, life, maxLife: life, type });
  }
  update() {
    for (let i = this.particles.length - 1; i >= 0; i--) {
      const p = this.particles[i];
      p.x += p.vx;
      p.y += p.vy;
      p.life--;
      if (p.type === 'smoke') {
        p.size = p.initialSize * (1 + (1 - p.life / p.maxLife) * 2.5);
        p.vx *= 0.94;
        p.vy *= 0.94;
      } else if (p.type === 'dirt') {
        p.vy += 0.35;
      }
      if (p.life <= 0) {
        this.particles.splice(i, 1);
      }
    }
  }
  draw(ctx, camX, camY) {
    for (const p of this.particles) {
      const alpha = Math.max(0, p.life / p.maxLife);
      ctx.save();
      ctx.globalAlpha = alpha;
      ctx.fillStyle = p.color;
      ctx.beginPath();
      ctx.arc(p.x - camX, p.y - camY, p.size, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }
  }
}

class Vehicle {
  constructor(x, y) {
    this.reset(x, y);
  }

  reset(x, y) {
    this.x = x;
    this.y = y;
    this.vx = 0;
    this.vy = 0;
    this.angle = 0;
    this.vAngle = 0;

    this.wheelDist = 48;
    this.wheelRadius = 18;
    this.suspensionRest = 24;
    
    this.rearWheel = { x: x - this.wheelDist, y: y + 20, vy: 0, contact: false };
    this.frontWheel = { x: x + this.wheelDist, y: y + 20, vy: 0, contact: false };

    this.wheelRotation = 0;
    this.fuel = 100;
    this.maxFuel = 100;
    this.isDead = false;
    this.deathReason = '';
  }

  update(terrain, controls, particles) {
    if (this.isDead) return;

    if (controls.gas) {
      this.fuel -= 0.085;
    } else {
      this.fuel -= 0.015;
    }

    if (this.fuel <= 0) {
      this.fuel = 0;
      controls.gas = false;
    }

    const driveDir = { x: Math.cos(this.angle), y: Math.sin(this.angle) };
    const torque = 0.38;

    if (controls.gas) {
      if (this.rearWheel.contact || this.frontWheel.contact) {
        this.vx += driveDir.x * torque;
        this.vy += driveDir.y * torque;
        this.vAngle -= 0.0035;
      } else {
        this.vAngle -= 0.022;
      }

      if (Math.random() < 0.6) {
        const exX = this.x - driveDir.x * 45 - Math.sin(this.angle) * 10;
        const exY = this.y - driveDir.y * 45 + Math.cos(this.angle) * 10;
        particles.add(
          exX, exY,
          -driveDir.x * 2.5 + (Math.random() - 0.5) * 1.5,
          -driveDir.y * 2.5 - Math.random() * 1.5,
          Math.random() > 0.4 ? 'rgba(75, 85, 99, 0.7)' : 'rgba(249, 115, 22, 0.9)',
          Math.random() * 4 + 3,
          24,
          'smoke'
        );
      }
    }

    if (controls.brake) {
      if (this.rearWheel.contact || this.frontWheel.contact) {
        this.vx -= driveDir.x * (torque * 0.75);
        this.vy -= driveDir.y * (torque * 0.75);
        this.vAngle += 0.0035;
      } else {
        this.vAngle += 0.022;
      }
    }

    this.vy += 0.24;
    this.vx *= 0.992;
    this.vy *= 0.992;
    this.vAngle *= 0.96;

    this.x += this.vx;
    this.y += this.vy;
    this.angle += this.vAngle;

    const cosA = Math.cos(this.angle);
    const sinA = Math.sin(this.angle);

    const rwTargetX = this.x - cosA * this.wheelDist + sinA * this.suspensionRest;
    const rwTargetY = this.y - sinA * this.wheelDist - cosA * (-this.suspensionRest);

    const fwTargetX = this.x + cosA * this.wheelDist + sinA * this.suspensionRest;
    const fwTargetY = this.y + sinA * this.wheelDist - cosA * (-this.suspensionRest);

    this.rearWheel.x = rwTargetX;
    this.rearWheel.y = rwTargetY;
    this.frontWheel.x = fwTargetX;
    this.frontWheel.y = fwTargetY;

    const rwGroundY = terrain.getHeight(this.rearWheel.x);
    const fwGroundY = terrain.getHeight(this.frontWheel.x);

    this.rearWheel.contact = false;
    this.frontWheel.contact = false;

    if (this.rearWheel.y + this.wheelRadius > rwGroundY) {
      this.rearWheel.contact = true;
      const pen = (this.rearWheel.y + this.wheelRadius) - rwGroundY;
      this.y -= pen * 0.45;
      this.vy *= 0.4;
      this.vAngle += 0.012 * (this.vx > 0 ? 1 : -1);

      if (Math.abs(this.vx) > 2.5 && Math.random() < 0.4) {
        particles.add(
          this.rearWheel.x, rwGroundY,
          -this.vx * 0.4 + (Math.random() - 0.5) * 3,
          -Math.random() * 4 - 2,
          '#78350f',
          Math.random() * 3 + 2,
          18,
          'dirt'
        );
      }
    }

    if (this.frontWheel.y + this.wheelRadius > fwGroundY) {
      this.frontWheel.contact = true;
      const pen = (this.frontWheel.y + this.wheelRadius) - fwGroundY;
      this.y -= pen * 0.45;
      this.vy *= 0.4;
      this.vAngle += 0.012 * (this.vx > 0 ? 1 : -1);
    }

    if (this.rearWheel.contact && this.frontWheel.contact) {
      const groundAngle = Math.atan2(fwGroundY - rwGroundY, this.frontWheel.x - this.rearWheel.x);
      let diff = groundAngle - this.angle;
      while (diff < -Math.PI) diff += Math.PI * 2;
      while (diff > Math.PI) diff -= Math.PI * 2;
      this.angle += diff * 0.18;
      this.vAngle *= 0.5;
    }

    this.wheelRotation += (this.vx * 0.08);

    const headY = this.y - cosA * 26;

      this.die('CRASHED!', 'The vehicle flipped over and the driver crashed into the slope.');
    }

    if (this.fuel <= 0 && Math.abs(this.vx) < 0.25) {
      this.die('OUT OF FUEL!', 'Your tank ran dry! Grab more Jerry cans along the course.');
    }
  }

    window.addEventListener('keydown', (e) => {

        this.controls.brake = true;
      }
      if (e.code === 'KeyR' && this.car.isDead) {
        this.restart();
      }

      if (e.code === 'KeyD' || e.code === 'ArrowRight' || e.code === 'KeyW' || e.code === 'ArrowUp') {
        this.controls.gas = false;
      }
      if (e.code === 'KeyA' || e.code === 'ArrowLeft' || e.code === 'KeyS' || e.code === 'ArrowDown') {
        this.controls.brake = false;
      }
    });

    const gasBtn = document.getElementById('gasBtn');
    const brakeBtn = document.getElementById('brakeBtn');

    const handlePress = (btn, key, val) => {
      btn.addEventListener('touchstart', (e) => { e.preventDefault(); sfx.init(); this.controls[key] = val; btn.classList.add('active'); });
      btn.addEventListener('touchend', (e) => { e.preventDefault(); this.controls[key] = !val; btn.classList.remove('active'); });
      btn.addEventListener('mousedown', (e) => { e.preventDefault(); sfx.init(); this.controls[key] = val; btn.classList.add('active'); });
      btn.addEventListener('mouseup', (e) => { e.preventDefault(); this.controls[key] = !val; btn.classList.remove('active'); });
      btn.addEventListener('mouseleave', () => { this.controls[key] = false; btn.classList.remove('active'); });
    };

    handlePress(gasBtn, 'gas', true);
    handlePress(brakeBtn, 'brake', true);

    document.getElementById('restartBtn').addEventListener('click', () => this.restart());
  }

  restart() {
    this.car.reset(120, 380);
    this.collectibles.reset();
    this.scoreCoins = 0;
    this.maxDistance = 0;
    this.gameOverModal.classList.remove('active');
  }

    this.car.update(this.terrain, this.controls, this.particles);
    this.collectibles.update(this.car.x);
    this.particles.update();

    const speedKmh = Math.abs(this.car.vx * 3.6);
    sfx.setEngineRPM(Math.min((speedKmh / 80) + (this.controls.gas ? 0.35 : 0), 1.5));

    for (const item of this.collectibles.items) {
      if (!item.collected) {
        const dx = item.x - this.car.x;
        const dy = item.y - this.car.y;

          item.collected = true;
          if (item.type === 'coin') {
            this.scoreCoins += 25;
            sfx.playCoin();
          } else if (item.type === 'fuel') {
            this.car.fuel = Math.min(this.car.fuel + 45, 100);
            sfx.playFuel();
          }
        }
      }
    }

    const currentDist = Math.max(0, Math.floor((this.car.x - 120) / 10));
    if (currentDist > this.maxDistance) this.maxDistance = currentDist;

    const targetCamX = this.car.x - VW * 0.32;
    const targetCamY = this.car.y - VH * 0.62;
    this.camX += (targetCamX - this.camX) * 0.1;
    this.camY += (targetCamY - this.camY) * 0.1;

    this.hudCoins.textContent = `${this.scoreCoins}`;
