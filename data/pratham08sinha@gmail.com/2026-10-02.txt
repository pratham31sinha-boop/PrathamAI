    const AudioEngine = {
      ctx: null,
      init() {
        if (!this.ctx) {
          const AudioContext = window.AudioContext || window.webkitAudioContext;

      beep(freq, duration, type = 'sine', sweep = 0) {
        if (!this.ctx) return;
        try {
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.type = type;
          osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
          if (sweep !== 0) {
            osc.frequency.exponentialRampToValueAtTime(Math.max(20, freq + sweep), this.ctx.currentTime + duration);
          }
          gain.gain.setValueAtTime(0.18, this.ctx.currentTime);
          gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
          osc.connect(gain);
          gain.connect(this.ctx.destination);
          osc.start();
          osc.stop(this.ctx.currentTime + duration);
        } catch(e) {}

        ctx.arc(p.x - cameraX, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
        ctx.globalAlpha = 1;
      });
    }

    function spawnDust(x, y, color = '#a7f3d0') {
      for (let i = 0; i < 5; i++) {
        particles.push({
          x: x,
          y: y,
          vx: (Math.random() - 0.5) * 3,
          vy: -Math.random() * 2,
          size: Math.random() * 4 + 2,
          color: color,
          alpha: 0.8,
          fade: 0.04
        });
      }
    }

    function update() {
      if (!isPlaying || isPaused) return;

      let moveDir = 0;
      if (keys.left) moveDir -= 1;
      if (keys.right) moveDir += 1;

      if (moveDir !== 0) {
        panda.facing = moveDir;
        panda.vx = moveDir * (panda.isRolling ? panda.speed * 1.6 : panda.speed);
        panda.animTime += 0.05;
      } else {
        panda.vx *= 0.78;
        if (Math.abs(panda.vx) < 0.1) panda.vx = 0;
      }

      if (keys.dash && !panda.isRolling && stamina >= 25) {
        panda.isRolling = true;
        panda.rollTimer = 22;
        stamina = Math.max(0, stamina - 28);
        AudioEngine.dash();
        spawnDust(panda.x + panda.w / 2, panda.y + panda.h, '#60a5fa');
      }

      if (panda.isRolling) {
        panda.rollTimer--;
        panda.animTime += 0.1;
        if (panda.rollTimer <= 0) {
          panda.isRolling = false;
        }
      }

      if (stamina < 100) stamina = Math.min(100, stamina + 0.25);
      document.getElementById('stamina-display').innerText = Math.round(stamina) + '%';

      panda.vy += 0.65;
      if (panda.vy > 14) panda.vy = 14;

      panda.x += panda.vx;
      panda.y += panda.vy;

      panda.isGrounded = false;
      platforms.forEach(p => {
        if (
          panda.x + panda.w > p.x &&
          panda.x < p.x + p.w &&
          panda.y + panda.h >= p.y &&
          panda.y + panda.h <= p.y + 24 &&
          panda.vy >= 0
        ) {
          panda.y = p.y - panda.h;
          panda.vy = 0;
          panda.isGrounded = true;
          panda.jumpsLeft = 2;
        }
      });

      if (keys.jump) {
        if (panda.jumpsLeft > 0) {
          panda.vy = panda.jumpForce;
          panda.jumpsLeft--;
          if (panda.jumpsLeft === 1) AudioEngine.jump();
          else AudioEngine.doubleJump();
          spawnDust(panda.x + panda.w / 2, panda.y + panda.h);
        }
        keys.jump = false;
      }

      const targetCamX = panda.x - W * 0.35;
      cameraX += (targetCamX - cameraX) * 0.1;
      if (cameraX < 0) cameraX = 0;

      const lastPlatform = platforms[platforms.length - 1];
      if (lastPlatform && lastPlatform.x + lastPlatform.w - cameraX < W * 2) {
        spawnWorldChunk(lastPlatform.x + lastPlatform.w, 2000);
      }

      items.forEach(it => {
        if (it.collected) return;
        if (
          Math.hypot(panda.x + panda.w / 2 - it.x, panda.y + panda.h / 2 - it.y) < 32
        ) {
          it.collected = true;
          if (it.type === 'bamboo') {
            score += 100;
            AudioEngine.pickup();
            spawnDust(it.x, it.y, '#facc15');
          } else {
            score += 250;
            stamina = Math.min(100, stamina + 40);
            AudioEngine.gem();
            spawnDust(it.x, it.y, '#38bdf8');
          }
          document.getElementById('score-display').innerText = score;
        }
      });

      if (panda.invulnerable > 0) {
        panda.invulnerable--;
      } else {
        obstacles.forEach(ob => {
          if (
            panda.x + panda.w > ob.x + 4 &&
            panda.x < ob.x + ob.w - 4 &&
            panda.y + panda.h > ob.y + 4 &&
            panda.y < ob.y + ob.h
          ) {
            if (panda.isRolling) {
              score += 150;
              AudioEngine.pickup();
              spawnDust(ob.x + ob.w / 2, ob.y + ob.h / 2, '#fbbf24');
              ob.y = 99999;
            } else {
              lives--;
              panda.invulnerable = 60;
              panda.vy = -8;
              panda.vx = -panda.facing * 6;
              AudioEngine.hit();
              document.getElementById('lives-display').innerText = lives;
              if (lives <= 0) {
                gameOver();
              }
            }
          }
        });
      }

      if (panda.y > H + 50) {
        lives--;
        document.getElementById('lives-display').innerText = lives;
        if (lives > 0) {
          panda.x = cameraX + 120;
          panda.y = groundY - 60;
          panda.vy = 0;
          panda.invulnerable = 60;
        } else {
          gameOver();
        }
      }
    }

    function render() {
      ctx.clearRect(0, 0, W, H);
      drawBackground();
      drawPlatforms();
      drawObstacles();
      drawItems();
      drawPanda();

      update();
      requestAnimationFrame(render);
    }
    requestAnimationFrame(render);

    function startGame() {
      AudioEngine.init();
      score = 0;
      lives = 3;
      stamina = 100;
      document.getElementById('score-display').innerText = '0';
      document.getElementById('lives-display').innerText = '3';
      document.getElementById('stamina-display').innerText = '100%';
      document.getElementById('game-modal').classList.remove('active');
      initLevel();
      isPlaying = true;
      isPaused = false;
    }

    function gameOver() {
      isPlaying = false;
      AudioEngine.gameOver();
      if (score > highScore) {
        highScore = score;
        localStorage.setItem('panda_valley_hi', highScore);
      }
      document.getElementById('modal-title').innerText = 'Run Complete!';
      document.getElementById('modal-sub').innerText = 'Bao Bao rested in the bamboo glade.';
      document.getElementById('final-score').innerText = score;
      document.getElementById('best-score').innerText = highScore;
      document.getElementById('modal-stats').style.display = 'flex';
      document.getElementById('start-btn').innerText = 'Play Again';
      document.getElementById('game-modal').classList.add('active');
    }

    function togglePause() {
      if (!isPlaying) return;
      isPaused = !isPaused;
      document.getElementById('pause-icon').innerText = isPaused ? '▶️' : '⏸️';
    }

    window.addEventListener('keydown', e => {
      AudioEngine.init();
      if (e.code === 'ArrowLeft' || e.code === 'KeyA') keys.left = true;
      if (e.code === 'ArrowRight' || e.code === 'KeyD') keys.right = true;
      if (e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'KeyW') keys.jump = true;
      if (e.code === 'ShiftLeft' || e.code === 'ShiftRight' || e.code === 'ArrowDown' || e.code === 'KeyS') keys.dash = true;
    });

    window.addEventListener('keyup', e => {
      if (e.code === 'ArrowLeft' || e.code === 'KeyA') keys.left = false;
      if (e.code === 'ArrowRight' || e.code === 'KeyD') keys.right = false;
      if (e.code === 'ShiftLeft' || e.code === 'ShiftRight' || e.code === 'ArrowDown' || e.code === 'KeyS') keys.dash = false;
    });

    function setupTouch(id, keyName) {
      const el = document.getElementById(id);
      if (!el) return;
      const activate = e => {
        e.preventDefault();
        AudioEngine.init();
        keys[keyName] = true;
        el.classList.add('active');
      };
      const deactivate = e => {
        e.preventDefault();
        keys[keyName] = false;
        el.classList.remove('active');
      };
      el.addEventListener('touchstart', activate, { passive: false });
      el.addEventListener('touchend', deactivate, { passive: false });
      el.addEventListener('touchcancel', deactivate, { passive: false });
      el.addEventListener('mousedown', activate);
      el.addEventListener('mouseup', deactivate);
      el.addEventListener('mouseleave', deactivate);
    }

    setupTouch('btn-left', 'left');
    setupTouch('btn-right', 'right');
    setupTouch('btn-jump', 'jump');
    setupTouch('btn-dash', 'dash');
  </script>
</body>
